        function getYouTubeEmbedId(url) {
            if (!url) return null;
            const regExp = /^.*(youtu.be\/|v\/|u\/\w\/|embed\/|watch\?v=|\&v=)([^#\&\?]*).*/;
            const match = url.match(regExp);
            return (match && match[2].length === 11) ? match[2] : null;
        }

        function openMaterialModal(editId = null) {
            document.getElementById('modal-material').classList.remove('hidden');
            if (editId) {
                const m = materials.find(item => item.id === editId);
                if (m) {
                    document.getElementById('modal-material-title').innerText = 'Edit Materi Pembelajaran';
                    document.getElementById('material-id').value = m.id;
                    document.getElementById('material-title-input').value = m.title;
                    document.getElementById('material-category-input').value = m.category;
                    document.getElementById('material-content-input').value = m.content;
                    document.getElementById('material-youtube-input').value = m.youtubeUrl || '';
                    document.getElementById('material-file-input').value = m.fileUrl || '';
                }
            } else {
                document.getElementById('modal-material-title').innerText = 'Tambah Materi Baru';
                document.getElementById('material-form').reset();
                document.getElementById('material-id').value = '';
            }
        }

        function closeMaterialModal() {
            document.getElementById('modal-material').classList.add('hidden');

        function handleMaterialSubmit(e) {
            e.preventDefault();
            const id = document.getElementById('material-id').value;
            const title = document.getElementById('material-title-input').value;
            const category = document.getElementById('material-category-input').value;
            const content = document.getElementById('material-content-input').value;
            const youtubeUrl = document.getElementById('material-youtube-input').value;
            const fileUrl = document.getElementById('material-file-input').value;

            if (id) {
                // Update
                const index = materials.findIndex(m => m.id === id);
                if (index !== -1) {
                    materials[index] = { ...materials[index], title, category, content, youtubeUrl, fileUrl };
                    showToast('Materi berhasil diperbarui', 'success');
                }
            } else {
                // Create
                const newMat = {
                    id: 'mat-' + Date.now(),
                    title,
                    category,
                    content,
                    youtubeUrl,
                    fileUrl,
                    teacherName: currentUser.name,
                    createdAt: getTodayDateString()
                };
                materials.unshift(newMat);
                showToast('Materi baru berhasil ditambahkan', 'success');
            }

            saveStore();
            closeMaterialModal();
            renderAllViews();
        }

        function editMaterial(id) {
            openMaterialModal(id);
        }

        function deleteMaterial(id) {
            if (window.confirm('Apakah Anda yakin ingin menghapus materi ini?')) {
                materials = materials.filter(m => m.id !== id);
                saveStore();
                renderAllViews();
                showToast('Materi berhasil dihapus', 'info');
            }
        }

                const myToday = attendance.find(a => a.studentId === currentUser.id && a.date === todayStr);
                const statusBox = document.getElementById('checkin-status-box');
                const form = document.getElementById('attendance-form');

                if (myToday) {
                    statusBox.classList.remove('hidden');
                    statusBox.className = `p-4 rounded-xl border text-xs space-y-1 ${
                        myToday.status === 'Hadir' ? 'bg-emerald-50 border-emerald-200 text-emerald-800' : 'bg-amber-50 border-amber-200 text-amber-800'
                    }`;
                    statusBox.innerHTML = `
                        <p class="font-bold"><i class="fa-solid fa-circle-check"></i> Absensi Sudah Tercatat Hari Ini</p>
                        <p>Status: <strong>${myToday.status}</strong> (${myToday.timestamp} WIB)</p>
                        ${myToday.note ? `<p>Catatan: ${myToday.note}</p>` : ''}
                    `;
                    form.classList.add('hidden');
                } else {
                    statusBox.classList.add('hidden');
                    form.classList.remove('hidden');
                }

                // Render Student History
                const myHistory = attendance.filter(a => a.studentId === currentUser.id);
                const tbody = document.getElementById('student-attendance-table-body');
                if (myHistory.length === 0) {
                    tbody.innerHTML = `<tr><td colspan="3" class="p-4 text-center text-slate-400">Belum ada riwayat absensi.</td></tr>`;
                } else {
                    tbody.innerHTML = myHistory.map(a => `
                        <tr>
                            <td class="p-3 font-medium">${a.date} <span class="text-slate-400">(${a.timestamp})</span></td>
                            <td class="p-3">
                                <span class="px-2 py-0.5 text-[10px] font-bold rounded-full ${
                                    a.status === 'Hadir' ? 'bg-emerald-100 text-emerald-700' : 'bg-amber-100 text-amber-700'
                                }">${a.status}</span>
                            </td>
                            <td class="p-3 text-slate-500">${a.note || '-'}</td>
                        </tr>
                    `).join('');
                }
            }
        }

        function handleAttendanceSubmit(e) {
            e.preventDefault();
            const status = document.querySelector('input[name="status"]:checked')?.value || 'Hadir';
            const note = document.getElementById('attendance-note').value;
            const todayStr = getTodayDateString();
            const nowTime = new Date().toLocaleTimeString('id-ID');

            const newAtt = {
                id: 'att-' + Date.now(),
                studentId: currentUser.id,
                studentName: currentUser.name,
                status,
                note,
                date: todayStr,
                timestamp: nowTime
            };

            attendance.unshift(newAtt);
            saveStore();
            renderAttendanceView();
            showToast('Presensi Anda berhasil terkirim!', 'success');
        }

        function renderAttendanceTable() {
            const filterDate = document.getElementById('filter-attendance-date')?.value;
            const filterStatus = document.getElementById('filter-attendance-status')?.value;

            let filtered = attendance;
            if (filterDate) {
                filtered = filtered.filter(a => a.date === filterDate);
            }
            if (filterStatus && filterStatus !== 'ALL') {
                filtered = filtered.filter(a => a.status === filterStatus);
            }

            const tbody = document.getElementById('teacher-attendance-table-body');
            if (filtered.length === 0) {
                tbody.innerHTML = `<tr><td colspan="4" class="p-4 text-center text-slate-400">Tidak ada data absensi sesuai filter.</td></tr>`;
                return;
            }

            tbody.innerHTML = filtered.map(a => `
                <tr class="hover:bg-slate-50">
                    <td class="p-3 font-bold text-slate-800">${a.studentName}</td>
                    <td class="p-3">${a.date} <span class="text-slate-400">(${a.timestamp})</span></td>
                    <td class="p-3">
                        <span class="px-2.5 py-1 text-[10px] font-bold rounded-full ${
                            a.status === 'Hadir' ? 'bg-emerald-100 text-emerald-700' :
                            a.status === 'Sakit' ? 'bg-rose-100 text-rose-700' : 'bg-amber-100 text-amber-700'
                        }">${a.status}</span>
                    </td>
                    <td class="p-3 text-slate-500">${a.note || '-'}</td>
                </tr>
            `).join('');
        }

        function exportAttendanceCSV() {
            if (attendance.length === 0) {
                showToast('Belum ada data absensi untuk diexport', 'warning');
                return;
            }

            let csvContent = "data:text/csv;charset=utf-8,Nama Siswa,Status,Tanggal,Jam,Catatan\n";
            attendance.forEach(a => {
                csvContent += `"${a.studentName}","${a.status}","${a.date}","${a.timestamp}","${a.note || ''}"\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `Absensi_beeducation_${getTodayDateString()}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);

            showToast('File CSV berhasil diunduh', 'success');
        }

                        </div>

                        <!-- Add Comment Form -->

                            <input type="text" id="comment-input-${post.id}" placeholder="Tulis tanggapan Anda..." class="flex-1 text-xs p-2.5 border border-slate-200 rounded-xl focus:ring-2 focus:ring-blue-500 focus:outline-none">
                            <button onclick="addComment('${post.id}')" class="bg-blue-600 hover:bg-blue-700 text-white px-4 py-2.5 rounded-xl text-xs font-bold transition">

        function handleAnnouncementSubmit(e) {
            e.preventDefault();
            const text = document.getElementById('announcement-text-input').value;
            const newAnc = {
                id: 'anc-' + Date.now(),
                authorName: currentUser.name,
                authorRole: currentUser.role,
                text,
                createdAt: new Date().toLocaleDateString('id-ID') + ' ' + new Date().toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' }),
                comments: []
            };

            announcements.unshift(newAnc);
            saveStore();
            closeAnnouncementModal();
            document.getElementById('announcement-form').reset();
            renderAllViews();
            showToast('Pengumuman baru diposting', 'success');
        }

        function addComment(postId) {
            const input = document.getElementById(`comment-input-${postId}`);
            const text = input?.value?.trim();
            if (!text) return;

            const post = announcements.find(a => a.id === postId);
            if (post) {
                if (!post.comments) post.comments = [];
                post.comments.push({
                    id: 'c-' + Date.now(),
                    name: currentUser.name,
                    role: currentUser.role,
                    text: text,
                    date: new Date().toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' })
                });
                saveStore();
                renderForumView();
                showToast('Tanggapan dikirim', 'success');
            }
        }

        function showToast(message, type = 'info') {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            const bgColor = type === 'success' ? 'bg-emerald-600' : type === 'warning' ? 'bg-amber-600' : 'bg-blue-600';
            
            toast.className = `${bgColor} text-white px-4 py-3 rounded-xl shadow-lg text-xs font-semibold flex items-center space-x-2 transition-all duration-300 transform translate-y-2 opacity-0`;
            toast.innerHTML = `
                <i class="fa-solid fa-circle-info"></i>
                <span>${message}</span>
            `;

            container.appendChild(toast);
            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
